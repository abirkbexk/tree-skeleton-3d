treeskel

treeskel - 3D apple-tree skeleton extraction from RGB-D PNG images (MMA3001 individual project).

Typical use::

from treeskel import get_config, train_pipeline, extract_skeleton_3d
out = train_pipeline(get_config(DATA_MODE="local", LOCAL_PATH="path/to/pngs"), out_dir="outputs")
res = extract_skeleton_3d("tree_rgb.png", "tree_depth.png", model=out["model"])

44    """Return a fresh configuration dict.
45
46    Args:
47        **overrides: any key of :data:`DEFAULT_CONFIG`. ``QUICK_TEST=True`` applies small, fast settings first
48            (explicit overrides still win).
49
50    Raises:

53    unknown = set(overrides) - set(DEFAULT_CONFIG)

61        cfg["GROUP_REGEX"] = REPO_GROUP_REGEX
62    return cfg

41    def predict(self, rgb, dnorm):
42        """Args: rgb (N,S,S,3) uint8, dnorm (N,S,S). Returns (rows (N,2,S) px after post-processing, extent (N,2) top/bottom rows)."""
43        S = self.cfg["IMG_SIZE"]
44        X = featurise(rgb, dnorm, self.cfg["FEAT_SIZE"])
45        if self.name == "CNN":
46            from .cnn import predict_cnn
47            rows = predict_cnn(self.model, rgb, dnorm, S)
48        else:
49            rows = predict_rows_sklearn(self.model, X, S)
50        ext = np.clip(self.extent_model.predict(X) * (S - 1), 0, S - 1)
51        return postprocess(rows, *self.post), ext

53    def save(self, path):
54        """Persist to one joblib file (CNN weights are stored as a CPU state_dict)."""
55        model = self.model
56        if self.name == "CNN":
57            model = {k: v.cpu() for k, v in self.model.state_dict().items()}
58        joblib.dump(dict(name=self.name, model=model, extent_model=self.extent_model, post=self.post, cfg=self.cfg), path)

def extract_skeleton_3d(rgb_path, depth_path=None, model=None, cfg=None, out_dir=None): 
131def extract_skeleton_3d(rgb_path, depth_path=None, model=None, cfg=None, out_dir=None):
132    """Estimate the 3D skeleton (trunk + two primary branches) of one apple tree from RGB(-D) PNG(s).
133
134    Args:
135        rgb_path: path to the RGB PNG. depth_path: optional aligned depth PNG (uint16, millimetres by default).
136        model: :class:`SkeletonModel` (or a path to a saved one). cfg: config dict (defaults to the model's own).
137        out_dir: if given, writes ``<stem>_skeleton3d.csv`` and ``.ply`` there (only when depth is usable).
138
139    Returns:
140        dict(rows_px (2,m), v (m,) image rows, points_3d {'left','right'} (m,3) metres or None, length_m {'left','right'} or None,
141        junction_row, extent (top, bottom), used_depth bool). 3D frame: X right, Y down, Z forward (metres).
142
143    Raises:
144        FileNotFoundError: missing RGB or depth file.  ValueError: input is not a PNG.
145    """
146    if model is None:
147        raise ValueError("a trained SkeletonModel (or path to one) is required")
148    if not isinstance(model, SkeletonModel):
149        model = SkeletonModel.load(model)
150    cfg = cfg or model.cfg; S = cfg["IMG_SIZE"]
151    rgb_path = Path(rgb_path)
152    if not rgb_path.exists():
153        raise FileNotFoundError(rgb_path)
154    if rgb_path.suffix.lower() != ".png":
155        raise ValueError("only PNG input is supported")
156    if depth_path is not None and not Path(depth_path).exists():
157        raise FileNotFoundError(depth_path)
158    s = load_sample(dict(rgb=rgb_path, depth=depth_path, skel=None), S, cfg, need_skeleton=False)
159    used_depth = s["depth"] is not None and np.isfinite(s["depth"]).mean() > 0.2
160    if not used_depth:
161        warnings.warn("no usable depth: returning 2D skeleton only; accuracy is reduced (model was trained with a depth channel)")
162    dn = depth_normalised(s["depth"]) if used_depth else np.zeros((S, S), np.float32)
163    rows, ext = model.predict(s["rgb"][None], dn[None].astype(np.float16)); rows, ext = rows[0], ext[0]
164    top, bot = int(round(ext[0])), int(round(ext[1]))
165    v = np.unique(np.arange(min(top, bot - 10), max(bot, top + 10) + 1).clip(0, S - 1))
166    jr = junction_row(rows[0][v], rows[1][v], cfg["JUNCTION_TOL_PX"])
167    res = dict(rows_px=rows[:, v], v=v, points_3d=None, length_m=None, junction_row=float(v[int(jr)]) if np.isfinite(jr) else np.nan,
168               extent=(int(v[0]), int(v[-1])), used_depth=bool(used_depth))
169    if used_depth:
170        pts = {b: lift_to_3d(rows[i][v], v.astype(float), s["depth"], s["scale"], cfg) for i, b in enumerate(("left", "right"))}
171        if all(p is not None for p in pts.values()):
172            res["points_3d"] = pts; res["length_m"] = {b: arc_length(v.astype(float), p, "simpson") for b, p in pts.items()}
173    if out_dir is not None and res["points_3d"] is not None:
174        out_dir = Path(out_dir); out_dir.mkdir(parents=True, exist_ok=True)
175        df = pd.concat([pd.DataFrame(p, columns=["X_m", "Y_m", "Z_m"]).assign(branch=b, row=v) for b, p in res["points_3d"].items()])
176        df.to_csv(out_dir / f"{rgb_path.stem}_skeleton3d.csv", index=False)
177        with open(out_dir / f"{rgb_path.stem}_skeleton3d.ply", "w") as fh:
178            fh.write(f"ply\nformat ascii 1.0\nelement vertex {len(df)}\nproperty float x\nproperty float y\nproperty float z\nend_header\n")
179            fh.writelines(f"{x:.5f} {y:.5f} {z:.5f}\n" for x, y, z in df[["X_m", "Y_m", "Z_m"]].values)
180    return res

 89    """Full training run. Model selection uses the validation set; the test set is only scored for the final report.
 90
 91    Returns:
 92        dict(model=SkeletonModel, ds, X, splits=(TR, VA, TE), results (DataFrame sorted by val RMSE), post_table, info, pred_val, pred_test).
 93    """
 94    S = cfg["IMG_SIZE"]
 95    root = acquire_data(cfg); records = discover_samples(root, cfg)
 96    ds = assemble_dataset(records, cfg); X = featurise(ds["rgb"], ds["dnorm"], cfg["FEAT_SIZE"])
 97    TR, VA, TE = make_splits(ds["group"], cfg["SEED"])
 98    if verbose: print(f"train {len(TR)} | val {len(VA)} | test {len(TE)}")
 99    fitted, info = train_classical(X, ds["targets"], ds["group"], TR, cfg)
100    pred_val, pred_test, table = {}, {}, []
101    for name, model in fitted.items():
102        pred_val[name], mva, _ = evaluate_model(model, X, VA, ds, cfg)
103        pred_test[name], mte, ms = evaluate_model(model, X, TE, ds, cfg)
104        table.append(dict(model=name, val_rmse_px=mva.rmse_px.mean(), test_rmse_px=mte.rmse_px.mean(), test_r=mte.r.mean(), test_chamfer_px=mte.chamfer_px.mean(),
105                          test_junction_px=mte.junction_px.mean(), fit_s=info[name]["fit_s"], predict_ms_per_img=ms))
106    if cfg["RUN_CNN"]:
107        from .cnn import predict_cnn, train_cnn

109        fitted["CNN"] = net; info["CNN"] = dict(best_params=dict(epochs=best_ep + 1), cv_rmse_px=np.nan, fit_s=fit_s, history=hist)
110        pred_val["CNN"] = predict_cnn(net, ds["rgb"][VA], ds["dnorm"][VA], S)
111        t0 = time.perf_counter(); pred_test["CNN"] = predict_cnn(net, ds["rgb"][TE], ds["dnorm"][TE], S); ms = 1000 * (time.perf_counter() - t0) / len(TE)
112        mva = evaluate_predictions(pred_val["CNN"], ds["rows"][VA], ds["valid"][VA], cfg["JUNCTION_TOL_PX"])
113        mte = evaluate_predictions(pred_test["CNN"], ds["rows"][TE], ds["valid"][TE], cfg["JUNCTION_TOL_PX"])
114        table.append(dict(model="CNN", val_rmse_px=mva.rmse_px.mean(), test_rmse_px=mte.rmse_px.mean(), test_r=mte.r.mean(), test_chamfer_px=mte.chamfer_px.mean(),
115                          test_junction_px=mte.junction_px.mean(), fit_s=fit_s, predict_ms_per_img=ms))
116    results = pd.DataFrame(table).set_index("model").round(3).sort_values("val_rmse_px")
117    best = results.index[0]
118    if verbose: print(results.to_string()); print("best model by validation RMSE:", best)
119    post_table, post = select_postprocessing(pred_val[best], ds, VA, cfg)
120    alpha = info["Ridge"]["best_params"]["reg__regressor__alpha"]
121    extent_model = fit_extent_model(X, ds, TR, cfg, alpha)
122    model = SkeletonModel(best, fitted[best], extent_model, post, cfg)
123    if out_dir is not None:
124        out = Path(out_dir); out.mkdir(parents=True, exist_ok=True)
125        results.to_csv(out / "model_comparison.csv"); post_table.to_csv(out / "postprocessing_selection.csv"); model.save(out / "skeleton_model.joblib")
126        if verbose: print("saved model + tables to", out)
127    return dict(model=model, ds=ds, X=X, splits=(TR, VA, TE), results=results, post_table=post_table, info=info, fitted=fitted,
128                pred_val=pred_val, pred_test=pred_test, post=post, alpha=alpha)
